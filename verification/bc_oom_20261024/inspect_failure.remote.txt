31: def limit_memory(gigabytes):
32:     """Cap trainer address space so a runaway read fails inside the trainer.
33: 
34:     The kernel then raises MemoryError, the trainer records its traceback and the
35:     coordinator reports a partial failure instead of an OOM kill of the run.

37:     if not gigabytes: return None

39:         import resource
40:         budget=int(float(gigabytes)*1024**3)

42:         if hard!=resource.RLIM_INFINITY: budget=min(budget,hard)
43:         resource.setrlimit(resource.RLIMIT_AS,(budget,budget))
44:         return budget
45:     except (ImportError,ValueError,OSError):
